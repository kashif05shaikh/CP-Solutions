#include <bits/stdc++.h>
using namespace std;
int main() {
    int T;
    cin >> T;
    while (T--) {
        int n;
        cin >> n;
        int cnt[3] = {0};
        for (int i = 0; i < n; i++) {
            int x;
            cin >> x;
            cnt[x]++;
        }
        int ans = cnt[0];
        int pairs = min(cnt[1], cnt[2]);
        ans += pairs;
        cnt[1] -= pairs;
        cnt[2] -= pairs;
        ans += cnt[1] / 3;
        ans += cnt[2] / 3;
        cout << ans << '\n';
    }
    return 0;
}
